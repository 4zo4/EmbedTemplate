set(ARCH_WIDTH 32 PARENT_SCOPE)
set(TARGET_CHIP_FLAGS
    -m32
    -march=i686
    -fno-builtin-printf
    -fno-builtin-puts
)
string(JOIN " " CPU_FLAGS ${TARGET_CHIP_FLAGS})
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} ${CPU_FLAGS}" PARENT_SCOPE)
set(CMAKE_ASM_FLAGS "${CMAKE_ASM_FLAGS} ${CPU_FLAGS}" PARENT_SCOPE)
set(LINKER_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/x86-virt.ld")
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -m32 -T${LINKER_SCRIPT} -Wl,--no-warn-rwx-segments" PARENT_SCOPE)
set(EXTRA_LIBS "-lgcc" PARENT_SCOPE)

set(FREERTOS_CHIP_DIR "${CMAKE_CURRENT_SOURCE_DIR}" PARENT_SCOPE)
set(FREERTOS_PORT "IA32_flat" PARENT_SCOPE)
set(TARGET_CHIP_SOURCES
    "${CMAKE_CURRENT_SOURCE_DIR}/common.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/port.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/startup.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/time.c"
)
if(NOT DEFINED ENABLE_PCI)
    set(ENABLE_PCI TRUE)
endif()
if(ENABLE_PCI)
    list(APPEND TARGET_CHIP_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/pci.c")
endif()
if(RTOS)
    list(APPEND TARGET_CHIP_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/io_apic.c")
else()
    list(APPEND TARGET_CHIP_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/apic.c")
endif()

set(TARGET_CHIP_SOURCES ${TARGET_CHIP_SOURCES} PARENT_SCOPE)
